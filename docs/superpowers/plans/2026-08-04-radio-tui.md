# radio-tui Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `radio-tui`, a terminal radio player scoped to Kosovo and Saudi Arabia stations, installable as a `radio` CLI command on Omarchy.

**Architecture:** A Textual TUI app fetches/caches the station list from the Radio Browser API (filtered to country codes `XK`/`SA`), and drives playback by launching `mpv` as a background subprocess controlled entirely over its JSON IPC socket (play/pause/volume/now-playing metadata).

**Tech Stack:** Python 3.11+, [Textual](https://textual.textualize.io/) for the UI, `requests` for HTTP, stdlib `tomllib`/`socket`/`subprocess` for config and mpv IPC, `mpv` as an external runtime dependency, `pytest` + `pytest-asyncio` for tests.

Reference spec: `docs/superpowers/specs/2026-08-04-radio-tui-design.md`

## Global Constraints

- Countries in scope: Kosovo (`XK`) and Saudi Arabia (`SA`) only — no other countries, no general "world" browsing.
- No favorites/presets and no recently-played history in v1.
- Playback goes through the external `mpv` binary via its JSON IPC socket — no Python audio-decoding library.
- Station data source is the Radio Browser API at `https://all.api.radio-browser.info` — no other station directory.
- Python >= 3.11 (the plan relies on stdlib `tomllib`).
- Runtime dependencies limited to `textual` and `requests`; dev/test adds `pytest` and `pytest-asyncio`.
- Config/cache live under XDG paths: `~/.config/radio-tui/config.toml`, `~/.cache/radio-tui/stations.json`.
- Distributed as a CLI (`pipx install .` → `radio` command) plus a `.desktop` entry for Walker — no GTK/Waybar UI.

```python
"""Data model for a single radio station."""
from __future__ import annotations

from dataclasses import dataclass, field


@dataclass
class Station:
    """A single radio station, normalized from the Radio Browser API."""

    stationuuid: str = ""
    name: str = "Unknown station"
    url: str = ""
    countrycode: str = ""
    tags: list[str] = field(default_factory=list)
    language: str = ""
    bitrate: int = 0
    available: bool = True

    @classmethod
    def from_api_dict(cls, data: dict) -> "Station":
        """Build a Station from one entry of the Radio Browser API response."""
        url = data.get("url_resolved") or data.get("url") or ""
        tags_raw = data.get("tags") or ""
        tags = [tag.strip() for tag in tags_raw.split(",") if tag.strip()]
        return cls(
            stationuuid=data.get("stationuuid", ""),
            name=data.get("name") or "Unknown station",
            url=url,
            countrycode=data.get("countrycode", ""),
            tags=tags,
            language=data.get("language", ""),
            bitrate=int(data.get("bitrate") or 0),
        )

    def to_cache_dict(self) -> dict:
        """Serialize to a plain dict for JSON caching."""
        return {
            "stationuuid": self.stationuuid,
            "name": self.name,
            "url": self.url,
            "countrycode": self.countrycode,
            "tags": self.tags,
            "language": self.language,
            "bitrate": self.bitrate,
        }

    @classmethod
    def from_cache_dict(cls, data: dict) -> "Station":
        """Rebuild a Station from a cached dict (see to_cache_dict)."""
        return cls(
            stationuuid=data["stationuuid"],
            name=data["name"],
            url=data["url"],
            countrycode=data["countrycode"],
            tags=data.get("tags", []),
            language=data.get("language", ""),
            bitrate=data.get("bitrate", 0),
        )
```

`src/radio_tui/radio_browser.py`:

```python
"""Radio Browser API client with a local JSON cache."""
from __future__ import annotations

import json
import time
from dataclasses import dataclass
from pathlib import Path

import requests

from radio_tui.models import Station

API_BASE = "https://all.api.radio-browser.info/json/stations/bycountrycodeexact"
USER_AGENT = "radio-tui/0.1"
CACHE_MAX_AGE_SECONDS = 24 * 60 * 60


class NoStationsAvailable(Exception):
    """Raised when stations can't be fetched and no local cache exists either."""


@dataclass
class StationsResult:
    """Result of get_stations(): the station list plus where it came from."""

    stations: list[Station]
    from_cache: bool
    cache_timestamp: float | None


def fetch_stations(countrycodes: list[str], timeout: float = 10.0) -> list[Station]:
    """Fetch stations for the given country codes from the Radio Browser API."""
    stations: list[Station] = []
    headers = {"User-Agent": USER_AGENT}
    for code in countrycodes:
        response = requests.get(f"{API_BASE}/{code}", headers=headers, timeout=timeout)
        response.raise_for_status()
        stations.extend(Station.from_api_dict(item) for item in response.json())
    return stations


def load_cache(cache_path: Path) -> tuple[list[Station], float] | None:
    """Load cached stations and their fetch timestamp, or None if no cache exists."""
    if not cache_path.exists():
        return None
    payload = json.loads(cache_path.read_text())
    stations = [Station.from_cache_dict(item) for item in payload["stations"]]
    return stations, payload["fetched_at"]


def save_cache(cache_path: Path, stations: list[Station], fetched_at: float) -> None:
    """Write stations and a fetch timestamp to the cache file."""
    cache_path.parent.mkdir(parents=True, exist_ok=True)
    payload = {
        "fetched_at": fetched_at,
        "stations": [station.to_cache_dict() for station in stations],
    }
    cache_path.write_text(json.dumps(payload))


def is_stale(fetched_at: float, now: float, max_age_seconds: int = CACHE_MAX_AGE_SECONDS) -> bool:
    """Return True if a cache fetched at `fetched_at` is older than max_age_seconds."""
    return (now - fetched_at) > max_age_seconds


def get_stations(
    cache_path: Path,
    countrycodes: list[str],
    force_refresh: bool = False,
) -> StationsResult:
    """Get stations, preferring a fresh cache, falling back to the API, then a stale cache."""
    cached = load_cache(cache_path)
    now = time.time()

    if not force_refresh and cached is not None:
        stations, fetched_at = cached
        if not is_stale(fetched_at, now):
            return StationsResult(stations, from_cache=True, cache_timestamp=fetched_at)

    try:
        stations = fetch_stations(countrycodes)
    except requests.RequestException:
        if cached is not None:
            stations, fetched_at = cached
            return StationsResult(stations, from_cache=True, cache_timestamp=fetched_at)
        raise NoStationsAvailable(
            "Could not reach the Radio Browser API and no local cache exists."
        )

    save_cache(cache_path, stations, now)
    return StationsResult(stations, from_cache=False, cache_timestamp=now)
```

```python
"""Small TOML-backed config file for persisted user settings."""
from __future__ import annotations

import tomllib
from dataclasses import dataclass
from pathlib import Path

DEFAULT_VOLUME = 70
DEFAULT_TAB = "XK"
DEFAULT_MPV_BINARY = "mpv"


@dataclass
class Config:
    volume: int = DEFAULT_VOLUME
    last_tab: str = DEFAULT_TAB
    mpv_binary: str = DEFAULT_MPV_BINARY


def load_config(config_path: Path) -> Config:
    """Load config from a TOML file, falling back to defaults for missing keys."""
    if not config_path.exists():
        return Config()
    data = tomllib.loads(config_path.read_text())
    return Config(
        volume=int(data.get("volume", DEFAULT_VOLUME)),
        last_tab=str(data.get("last_tab", DEFAULT_TAB)),
        mpv_binary=str(data.get("mpv_binary", DEFAULT_MPV_BINARY)),
    )


def save_config(config_path: Path, config: Config) -> None:
    """Write config to a TOML file (flat key = value pairs, one per line)."""
    config_path.parent.mkdir(parents=True, exist_ok=True)
    lines = [
        f"volume = {config.volume}",
        f'last_tab = "{config.last_tab}"',
        f'mpv_binary = "{config.mpv_binary}"',
    ]
    config_path.write_text("\n".join(lines) + "\n")
```

```python
"""Low-level JSON-line protocol for talking to mpv's IPC socket."""
from __future__ import annotations

import json
import socket
from collections import deque


class MPVIPCClient:
    """Wraps a connected unix socket to mpv's --input-ipc-server."""

    def __init__(self, sock: socket.socket, max_events: int = 50):
        self._sock = sock
        self._buffer = b""
        self.events: deque[dict] = deque(maxlen=max_events)

    def send_command(self, command: list) -> dict:
        """Send one command and return mpv's response dict (the non-event line)."""
        self._sock.sendall(json.dumps({"command": command}).encode() + b"\n")
        while True:
            message = self._read_message()
            if "event" in message:
                self.events.append(message)
                continue
            return message

    def _read_message(self) -> dict:
        while b"\n" not in self._buffer:
            chunk = self._sock.recv(4096)
            if not chunk:
                raise ConnectionError("mpv IPC socket closed unexpectedly")
            self._buffer += chunk
        line, self._buffer = self._buffer.split(b"\n", 1)
        return json.loads(line)

    def close(self) -> None:
        self._sock.close()
```

```python
"""Manage an mpv subprocess as a background radio playback engine."""
from __future__ import annotations

import shutil
import socket
import subprocess
import tempfile
import time
from pathlib import Path

from radio_tui.mpv_ipc import MPVIPCClient

CONNECT_TIMEOUT_SECONDS = 2.0
CONNECT_RETRY_INTERVAL = 0.05


class MPVController:
    """Starts mpv in idle mode and drives it over its JSON IPC socket."""

    def __init__(self, mpv_binary: str = "mpv"):
        self._mpv_binary = mpv_binary
        self._process: subprocess.Popen | None = None
        self._socket_path: Path | None = None
        self._client: MPVIPCClient | None = None

    @staticmethod
    def is_installed(mpv_binary: str = "mpv") -> bool:
        return shutil.which(mpv_binary) is not None

    def start(self) -> None:
        """Launch mpv in idle mode and connect to its IPC socket."""
        self._socket_path = Path(tempfile.mktemp(prefix="radio-tui-mpv-", suffix=".sock"))
        self._process = subprocess.Popen(
            [
                self._mpv_binary,
                "--no-video",
                "--idle=yes",
                f"--input-ipc-server={self._socket_path}",
            ],
            stdout=subprocess.DEVNULL,
            stderr=subprocess.DEVNULL,
        )
        sock = self._connect_with_retry()
        self._client = MPVIPCClient(sock)

    def _connect_with_retry(self) -> socket.socket:
        deadline = time.monotonic() + CONNECT_TIMEOUT_SECONDS
        last_error: OSError | None = None
        while time.monotonic() < deadline:
            try:
                sock = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
                sock.connect(str(self._socket_path))
                return sock
            except OSError as error:
                last_error = error
                time.sleep(CONNECT_RETRY_INTERVAL)
        raise ConnectionError(f"Could not connect to mpv IPC socket: {last_error}")

    def play(self, url: str) -> None:
        self._client.send_command(["loadfile", url, "replace"])

    def toggle_pause(self) -> bool:
        current = self._client.send_command(["get_property", "pause"])["data"]
        new_state = not current
        self._client.send_command(["set_property", "pause", new_state])
        return new_state

    def set_volume(self, volume: int) -> None:
        volume = max(0, min(100, volume))
        self._client.send_command(["set_property", "volume", volume])

    def get_now_playing(self) -> str | None:
        response = self._client.send_command(["get_property", "media-title"])
        if response.get("error") != "success":
            return None
        return response.get("data")

    def had_playback_error(self) -> bool:
        """Return True and clear the flag if an end-file error occurred since the last check."""
        found = any(
            event.get("event") == "end-file" and event.get("reason") == "error"
            for event in self._client.events
        )
        if found:
            self._client.events.clear()
        return found

    def quit(self) -> None:
        if self._client is not None:
            try:
                self._client.send_command(["quit"])
            except ConnectionError:
                pass
            self._client.close()
        if self._process is not None:
            self._process.terminate()
            self._process.wait(timeout=2)
```

```python
"""Textual TUI application for browsing and playing radio stations."""
from __future__ import annotations

from pathlib import Path

from textual import work
from textual.app import App, ComposeResult
from textual.containers import Vertical
from textual.widgets import DataTable, Footer, Header, Input, TabbedContent, TabPane

from radio_tui import radio_browser
from radio_tui.models import Station
from radio_tui.radio_browser import NoStationsAvailable

CACHE_PATH = Path.home() / ".cache" / "radio-tui" / "stations.json"
COUNTRIES = [("XK", "Kosovo"), ("SA", "Saudi Arabia")]


class RadioApp(App):
    """Browse and play radio stations from Kosovo and Saudi Arabia."""

    BINDINGS = [("r", "refresh_stations", "Refresh")]

    def __init__(self, cache_path: Path = CACHE_PATH):
        super().__init__()
        self._cache_path = cache_path
        self._stations_by_country: dict[str, list[Station]] = {code: [] for code, _ in COUNTRIES}

    def compose(self) -> ComposeResult:
        yield Header()
        with TabbedContent():
            for code, label in COUNTRIES:
                with TabPane(label, id=f"tab-{code}"):
                    with Vertical():
                        yield Input(placeholder="Search…", id=f"search-{code}")
                        yield DataTable(id=f"table-{code}")
        yield Footer()

    def on_mount(self) -> None:
        for code, _ in COUNTRIES:
            table = self.query_one(f"#table-{code}", DataTable)
            table.add_columns("Name", "Tags", "Bitrate")
            table.cursor_type = "row"
        self.load_stations()

    @work(exclusive=True)
    async def load_stations(self, force_refresh: bool = False) -> None:
        try:
            result = radio_browser.get_stations(
                self._cache_path,
                [code for code, _ in COUNTRIES],
                force_refresh=force_refresh,
            )
        except NoStationsAvailable as error:
            self.exit(return_code=1, message=str(error))
            return

        for code, _ in COUNTRIES:
            self._stations_by_country[code] = [
                station for station in result.stations if station.countrycode == code
            ]
        self._refresh_tables()

    def _refresh_tables(self) -> None:
        for code, _ in COUNTRIES:
            self._render_table(code, self._stations_by_country[code])

    def _render_table(self, code: str, stations: list[Station]) -> None:
        table = self.query_one(f"#table-{code}", DataTable)
        table.clear()
        for station in stations:
            tags = ", ".join(station.tags)
            table.add_row(station.name, tags, str(station.bitrate), key=station.stationuuid)

    def on_input_changed(self, event: Input.Changed) -> None:
        code = event.input.id.removeprefix("search-")
        query = event.value.strip().lower()
        stations = self._stations_by_country[code]
        if query:
            stations = [
                station
                for station in stations
                if query in station.name.lower() or any(query in tag.lower() for tag in station.tags)
            ]
        self._render_table(code, stations)

    def action_refresh_stations(self) -> None:
        self.load_stations(force_refresh=True)
```

```python
from __future__ import annotations

from pathlib import Path

from textual import work
from textual.app import App, ComposeResult
from textual.containers import Vertical
from textual.widgets import DataTable, Footer, Header, Input, Static, TabbedContent, TabPane

from radio_tui import config as config_module
from radio_tui import radio_browser
from radio_tui.models import Station
from radio_tui.mpv_controller import MPVController
from radio_tui.radio_browser import NoStationsAvailable

CACHE_PATH = Path.home() / ".cache" / "radio-tui" / "stations.json"
CONFIG_PATH = Path.home() / ".config" / "radio-tui" / "config.toml"
COUNTRIES = [("XK", "Kosovo"), ("SA", "Saudi Arabia")]
VOLUME_STEP = 5


class RadioApp(App):
    """Browse and play radio stations from Kosovo and Saudi Arabia."""

    BINDINGS = [
        ("r", "refresh_stations", "Refresh"),
        ("space", "toggle_pause", "Pause/Resume"),
        ("plus", "volume_up", "Vol+"),
        ("right", "volume_up", "Vol+"),
        ("minus", "volume_down", "Vol-"),
        ("left", "volume_down", "Vol-"),
        ("q", "quit", "Quit"),
    ]

    def __init__(self, cache_path: Path = CACHE_PATH, config_path: Path = CONFIG_PATH, mpv=None):
        super().__init__()
        self._cache_path = cache_path
        self._config_path = config_path
        self._mpv_override = mpv
        self._stations_by_country: dict[str, list[Station]] = {code: [] for code, _ in COUNTRIES}
        self._current_station: Station | None = None

    def compose(self) -> ComposeResult:
        yield Header()
        with TabbedContent():
            for code, label in COUNTRIES:
                with TabPane(label, id=f"tab-{code}"):
                    with Vertical():
                        yield Input(placeholder="Search…", id=f"search-{code}")
                        yield DataTable(id=f"table-{code}")
        yield Static("", id="now-playing")
        yield Footer()

    def on_mount(self) -> None:
        self._config = config_module.load_config(self._config_path)
        self.mpv = self._mpv_override or MPVController(mpv_binary=self._config.mpv_binary)
        self.mpv.start()

        for code, _ in COUNTRIES:
            table = self.query_one(f"#table-{code}", DataTable)
            table.add_columns("Name", "Tags", "Bitrate")
            table.cursor_type = "row"
        self.load_stations()
        self.set_interval(2.0, self._poll_now_playing)

    @work(exclusive=True)
    async def load_stations(self, force_refresh: bool = False) -> None:
        try:
            result = radio_browser.get_stations(
                self._cache_path,
                [code for code, _ in COUNTRIES],
                force_refresh=force_refresh,
            )
        except NoStationsAvailable as error:
            self.exit(return_code=1, message=str(error))
            return

        for code, _ in COUNTRIES:
            self._stations_by_country[code] = [
                station for station in result.stations if station.countrycode == code
            ]
        self._refresh_tables()

    def _refresh_tables(self) -> None:
        for code, _ in COUNTRIES:
            self._render_table(code, self._stations_by_country[code])

    def _render_table(self, code: str, stations: list[Station]) -> None:
        table = self.query_one(f"#table-{code}", DataTable)
        table.clear()
        for station in stations:
            name = f"⚠ {station.name}" if not station.available else station.name
            tags = ", ".join(station.tags)
            table.add_row(name, tags, str(station.bitrate), key=station.stationuuid)

    def on_input_changed(self, event: Input.Changed) -> None:
        code = event.input.id.removeprefix("search-")
        query = event.value.strip().lower()
        stations = self._stations_by_country[code]
        if query:
            stations = [
                station
                for station in stations
                if query in station.name.lower() or any(query in tag.lower() for tag in station.tags)
            ]
        self._render_table(code, stations)

    def on_data_table_row_selected(self, event: DataTable.RowSelected) -> None:
        code = event.data_table.id.removeprefix("table-")
        station = next(
            (s for s in self._stations_by_country[code] if s.stationuuid == event.row_key.value),
            None,
        )
        if station is not None:
            self._play(station)

    def _play(self, station: Station) -> None:
        self._current_station = station
        self.mpv.play(station.url)
        self._update_now_playing(title=None)

    def _update_now_playing(self, title: str | None) -> None:
        if self._current_station is None:
            self.query_one("#now-playing", Static).update("")
            return
        text = f"♪ Now playing: {self._current_station.name}"
        if title:
            text += f" — {title}"
        text += f"  (vol {self._config.volume}%)"
        self.query_one("#now-playing", Static).update(text)

    def _poll_now_playing(self) -> None:
        if self._current_station is None:
            return
        if self.mpv.had_playback_error():
            self._current_station.available = False
            self._refresh_tables()
            return
        title = self.mpv.get_now_playing()
        self._update_now_playing(title)

    def action_refresh_stations(self) -> None:
        self.load_stations(force_refresh=True)

    def action_toggle_pause(self) -> None:
        self.mpv.toggle_pause()

    def action_volume_up(self) -> None:
        self._config.volume = min(100, self._config.volume + VOLUME_STEP)
        self.mpv.set_volume(self._config.volume)
        self._update_now_playing(title=None)

    def action_volume_down(self) -> None:
        self._config.volume = max(0, self._config.volume - VOLUME_STEP)
        self.mpv.set_volume(self._config.volume)
        self._update_now_playing(title=None)

    def action_quit(self) -> None:
        config_module.save_config(self._config_path, self._config)
        self.mpv.quit()
        self.exit()
```

```python
"""Entry point for the radio-tui CLI."""
from __future__ import annotations

import sys

from radio_tui.app import RadioApp
from radio_tui.mpv_controller import MPVController


def main() -> int:
    """CLI entry point. Returns a process exit code."""
    if not MPVController.is_installed():
        print("mpv is not installed. Install it with: pacman -S mpv", file=sys.stderr)
        return 1

    app = RadioApp()
    app.run()
    return app.return_code or 0


if __name__ == "__main__":
    sys.exit(main())
```

```markdown
# radio-tui

A terminal radio player for Kosovo and Saudi Arabia stations, built on
[Textual](https://textual.textualize.io/) and [mpv](https://mpv.io/).
```
